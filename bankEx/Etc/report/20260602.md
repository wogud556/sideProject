    USER_ID     VARCHAR2(50)    NOT NULL,

    PHONE       VARCHAR2(20),

);

-- 2. 계좌 정보
CREATE TABLE ACCOUNT_INFO (
    ACCOUNT_ID      NUMBER          GENERATED BY DEFAULT AS IDENTITY,

Oracle 사용 시 추가 확인사항

Oracle XE 기준으로 application-prod.yml에 아래처럼 JPA 설정을

spring:
  datasource:
    driver-class-name: oracle.jdbc.OracleDriver
    url: jdbc:oracle:thin:@localhost:1521/XE
    username: hanati
    password: (비밀번호)
  jpa:
    hibernate:
      ddl-auto: none        # 테이블을 직접 만들었으므로 none
    show-sql: true
    database-platform: org.hibernate.dialect.OracleDialect
  mybatis:
    mapper-locations: classpath:mapper/**/*.xml

▎ ddl-auto: none 으로 설정하면 Hibernate가 테이블을 건드리지 않고 위 스크립트로 만든 테이블을 그대로 사용합니다.
